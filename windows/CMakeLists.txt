cmake_minimum_required(VERSION 3.16)
project(DozorWindows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    add_executable(dozor-win dozor.cpp)
    target_link_libraries(dozor-win PRIVATE fwpuclnt iphlpapi bcrypt ole32)
    option(DOZOR_WINDOWS_INTEGRATION_TESTS "Build a test-only controller permitting loopback" OFF)
    if(DOZOR_WINDOWS_INTEGRATION_TESTS)
        add_executable(dozor-loopback-fixture dozor.cpp)
        target_compile_definitions(dozor-loopback-fixture PRIVATE DOZOR_TEST_LOOPBACK)
        target_link_libraries(dozor-loopback-fixture PRIVATE fwpuclnt iphlpapi bcrypt ole32 ws2_32)
    endif()
endif()
add_executable(dozor-policy-tests tests/policy.cpp)
enable_testing()
add_test(NAME policy COMMAND dozor-policy-tests)
